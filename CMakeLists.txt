cmake_minimum_required(VERSION 3.20)
project(death_improvements LANGUAGES C CXX)

if(NOT OpenRACMod_DIR)
  if(DEFINED OPENRAC_SDK)
    set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
  elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/sdk.lock")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk.lock" lock)
    string(JSON sdk_version GET "${lock}" version)
    set(OpenRACMod_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/openrac-mod-sdk-${sdk_version}/cmake")
    if(NOT EXISTS "${OpenRACMod_DIR}/OpenRACModConfig.cmake")
      message(FATAL_ERROR "No OpenRAC mod SDK ${sdk_version}: run python3 tools/fetch_sdk.py first")
    endif()
  endif()
endif()
find_package(OpenRACMod 1.0 REQUIRED)
include(CTest)

openrac_add_mod(death_improvements MANIFEST openrac-mod.json SOURCES src/mod.cpp)
target_include_directories(death_improvements PRIVATE src)
set_target_properties(death_improvements PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
if(MSVC)
  target_compile_options(death_improvements PRIVATE /utf-8)
else()
  target_compile_options(death_improvements PRIVATE -Wall -Wextra)
endif()

if(BUILD_TESTING)
  add_executable(death_improvements_logic_test tests/logic_test.cpp)
  target_include_directories(death_improvements_logic_test PRIVATE src)
  set_target_properties(death_improvements_logic_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  add_test(NAME death_improvements_logic COMMAND death_improvements_logic_test)
endif()

get_target_property(mod_package death_improvements OPENRAC_MOD_PACKAGE_DIR)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "${mod_package}/LICENSE" COPYONLY)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/README.md" "${mod_package}/README.md" COPYONLY)
